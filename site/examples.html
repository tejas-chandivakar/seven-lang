Examples

Learn SEVEN by example. Each snippet is a complete, runnable program.

Hello World

hello.seven 
show "Hello, World!"

Variables & Input

input.seven 
name = ask "What's your name? "
age = ask "How old are you? "

show "Hello, " + name + "!"
show "You are " + age + " years old."

Conditionals

conditions.seven 
score = 85

if score atleast 90
    show "Grade: A"
elif score atleast 80
    show "Grade: B"
elif score atleast 70
    show "Grade: C"
else
    show "Grade: F"
end

Loops

loops.seven 
!! Repeat N times
repeat 5 times as i
    show "Iteration " + i
end

!! While loop
count = 0
while count below 3
    show "Count: " + count
    count = count + 1
end

!! Each loop with index
fruits = ["apple", "banana", "cherry"]
each fruits as i, fruit
    show i + ": " + fruit
end

Functions

functions.seven 
func fibonacci(n)
    if n below 2
        return n
    end
    return fibonacci(n - 1) + fibonacci(n - 2)
end

repeat 10 times as i
    show "fib(" + i + ") = " + fibonacci(i)
end

Arrays

nums = [5, 2, 8, 1, 9, 3]

FizzBuzz

fizzbuzz.seven 
i = 1
while i atmost 100
    if i % 15 is 0
        show "FizzBuzz"
    elif i % 3 is 0
        show "Fizz"
    elif i % 5 is 0
        show "Buzz"
    else
        show i
    end
    i = i + 1
end

File I/O

files.seven 
!! Write, read, and manipulate files
write "todo.txt", "Buy groceries"
append("todo.txt", "\nClean house")
append("todo.txt", "\nWrite code")

tasks = lines("todo.txt")
each tasks as i, task
    show (i + 1) + ". " + task
end

delete("todo.txt")

Error Handling

errors.seven 
func safe_divide(a, b)
    if b is 0
        error "Division by zero!"
    end
    return a / b
end

try
    show safe_divide(10, 0)
catch e
    err "Error: " + e
done

OS & System

system.seven 
!! Timing
start = clock()
sleep(1)
elapsed = clock() - start
show "Elapsed: " + elapsed

!! Environment
home = env("HOME")
show "Home: " + home

!! Command-line args
argv = args()
each argv as arg
    show "Arg: " + arg
end

!! Run shell commands
system("echo Hello from the shell")

Natural Mode

English-like array syntax with mode natural.

natural.seven 
mode natural

colors = ["red", "blue"]
add "green" to colors
add "yellow" to colors

each colors as color
    show color
end
